29T ipow(T base, long long exp) {
30 T res = _one<T>();
31 while (exp) {
32 if (exp & 1LL) res *= base;

66template<class T>
67struct Comb {
68 std::vector<T> fact, ifact;
69 Comb(int n = 0) { if (n) init(n); }
70 void init(int n) {
71 fact.resize(n + 1);
72 ifact.resize(n + 1);
73 fact[0] = _one<T>();
74 for (int i = 1; i <= n; i++) fact[i] = fact[i - 1] * T(i);
75 ifact[n] = _one<T>() / fact[n];
76 for (int i = n; i >= 1; i--) ifact[i - 1] = ifact[i] * T(i);
77 }
78 T operator()(int n, int k) const {
79 if (k < 0 || k > n) return _zero<T>();
80 return fact[n] * ifact[k] * ifact[n - k];
81 }
82};

88// Binomial PMF: P(X=k)
89template<class T>
90T binom_pmf(int n, int k, T p, const Comb<T>& C) {
91 if (k < 0 || k > n) return _zero<T>();
92 return C(n, k) * ipow(p, k) * ipow(_one<T>() - p, n - k);
93}

95// Binomial CDF: P(X<=k)
96template<class T>

98 T s = _zero<T>();
99 for (int i = 0; i <= k; i++) s += binom_pmf(n, i, p, C);
100 return s;

104template<class T>

106 if (k < 0 || k > n || k > K || n - k > N - K) return _zero<T>();
